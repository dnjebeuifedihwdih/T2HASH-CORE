شما به ۲ سرور لینوکس (ترجیحاً Ubuntu 22.04 یا 24.04 با دسترسی root) نیاز دارید:

  • سرور خارج (Exit Node): ترافیک آزاد اینترنت را دریافت و به فیلترشکن تحویل می‌دهد.

Description=mh.tunnel Anti-Censorship Relay (Iran Client)
After=network.target network-online.target
Wants=network-online.target

[Service]
Type=simple
User=root
ExecStart=/usr/local/bin/t2tunnel -mode client -transport tls -listen 0.0.0.0:9000 -remote 13.140.81.175:4443 -key "mh_9f482a17cb03da5e718b45620f3e8291" -sni speedtest.net -utls -frag 4

RestartSec=2s
LimitNOFILE=1000000

[Install]
